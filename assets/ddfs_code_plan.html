def rk4_step(f, x, u, dt): 
"""
Single RK4 integration step.

Args:
    f: continuous dynamics function f(x, u) -> x_dot
    x: current state (n,)
    u: control input (m,)  [held constant over dt]
    dt: time step

Returns:
    x_next: state at t + dt
"""
k1 = f(x, u)
k2 = f(x + 0.5*dt*k1, u)
k3 = f(x + 0.5*dt*k2, u)
k4 = f(x + dt*k3, u)
return x + (dt/6)*(k1 + 2*k2 + 2*k3 + k4)

def numerical_jacobian(f_discrete, x, u, eps=1e-6): 
"""
Compute A = ∂f/∂x and B = ∂f/∂u at (x, u).

Args:
    f_discrete: x_next = f(x, u)
    x: linearization point state (n,)
    u: linearization point input (m,)
    eps: finite difference step

Returns:
    A: (n, n) Jacobian w.r.t. state
    B: (n, m) Jacobian w.r.t. input
"""
n, m = len(x), len(u)
f0 = f_discrete(x, u)

A = np.zeros((n, n))
for i in range(n):
    x_plus = x.copy(); x_plus[i] += eps
    x_minus = x.copy(); x_minus[i] -= eps
    A[:, i] = (f_discrete(x_plus, u) - f_discrete(x_minus, u)) / (2*eps)

B = np.zeros((n, m))
for j in range(m):
    u_plus = u.copy(); u_plus[j] += eps
    u_minus = u.copy(); u_minus[j] -= eps
    B[:, j] = (f_discrete(x, u_plus) - f_discrete(x, u_minus)) / (2*eps)

return A, B

class SegmentDataCollector: 
def __init__(self, T, L, n, m):
    """
    T: segment length (100)
    L: data window length (60)
    n: state dimension (4)
    m: input dimension (2)
    """
    self.T = T
    self.L = L
    self.n = n
    self.m = m
    
def get_segment_indices(self, i):
    """Return T_i = {k_i, ..., k_{i+1}-1}"""
    k_i = i * self.T
    return range(k_i, k_i + self.T)

def get_data_window_indices(self, i):
    """Return T_i^D = {k_{i+1}-L, ..., k_{i+1}-1}"""
    k_i_plus_1 = (i + 1) * self.T
    k_i_D = k_i_plus_1 - self.L
    return range(k_i_D, k_i_plus_1)

def is_in_data_window(self, k, i):
    """Check if k ∈ T_i^D"""
    return k in self.get_data_window_indices(i)

def build_data_matrices(self, eta_history, xi_history, i):
    """
    Build H_i, H_i^+, Ξ_i from logged data.
    
    Args:
        eta_history: dict {k: η(k)}
        xi_history: dict {k: ξ(k)}
        i: segment index
        
    Returns:
        H_i: (n, L)
        H_i_plus: (n, L)
        Xi_i: (m, L)
    """
    window = list(self.get_data_window_indices(i))
    
    H_i = np.column_stack([eta_history[k] for k in window])
    H_i_plus = np.column_stack([eta_history[k+1] for k in window])
    Xi_i = np.column_stack([xi_history[k] for k in window])

    """

8️⃣ Module: data_driven/beta_computation.py

Purpose

Compute the disturbance energy bound β_i (Eq. 16).

β_i = Σ_{k ∈ T_i^D} (C|k-k_i| ||[η(k); ξ(k)]|| + γ + L_r ||[η(k); ξ(k)]||²)²
def compute_beta(eta_history, xi_history, data_window, k_i, C, gamma, L_r): 
"""
Compute β_i from Eq. 16.

Args:
    eta_history: dict of η(k) values
    xi_history: dict of ξ(k) values
    data_window: indices in T_i^D
    k_i: start of segment i
    C, gamma, L_r: Lipschitz constants
    
Returns:
    beta_i: scalar bound
"""
beta_i = 0.0

for k in data_window:
    eta_k = eta_history[k]
    xi_k = xi_history[k]
    z_k = np.concatenate([eta_k, xi_k])
    norm_z = np.linalg.norm(z_k)
    
    term = C * abs(k - k_i) * norm_z + gamma + L_r * norm_z**2
    beta_i += term**2

return beta_i

9️⃣ Module: synthesis/ellipsoid_bounds.py

Purpose

Compute P_min(k), R_max(k) via maximum-volume inscribed ellipsoid (MVIE).

State Constraint MVIE (Remark 2)

P_min(k)^{-1/2} = argmax_{Z≻0} log det(Z) s.t. ||Z a_j^x(k)||_2 ≤ b_j^x(k) ∀j 0 ⪯ Z ⪯ x_max I
def compute_mvie_state(x_nom_k, x_bounds): 
"""
Compute P_min(k) for state constraints.

Args:
    x_nom_k: nominal state at time k (4,)
    x_bounds: dict with 'lower' and 'upper' arrays (4,) each
    
Returns:
    P_min_k: (4, 4) positive definite matrix
"""
n = 4

# Deviation constraints: -x_nom + x_lower ≤ η ≤ x_upper - x_nom
# In form a_j^T η ≤ b_j:
#   η_i ≤ x_upper_i - x_nom_i  =>  e_i^T η ≤ (x_upper - x_nom)_i
#  -η_i ≤ x_nom_i - x_lower_i  => -e_i^T η ≤ (x_nom - x_lower)_i

A_ineq = np.vstack([np.eye(n), -np.eye(n)])  # (2n, n)
b_ineq = np.concatenate([
    x_bounds['upper'] - x_nom_k,
    x_nom_k - x_bounds['lower']
])  # (2n,)

# Solve MVIE via CVXPY
Z = cp.Variable((n, n), symmetric=True)
constraints = [Z >> 0]

for i in range(2*n):
    a_i = A_ineq[i]
    b_i = b_ineq[i]
    constraints.append(cp.norm(Z @ a_i, 2) <= b_i)

prob = cp.Problem(cp.Maximize(cp.log_det(Z)), constraints)
prob.solve(solver=cp.SCS)

Z_opt = Z.value
P_min_k = np.linalg.inv(Z_opt @ Z_opt)  # P = Z^{-2}

return P_min_k

🔟 Module: synthesis/lmi_matrices.py

Purpose

Construct the LMI matrices S, Ñ₁, Ñ₂ from Theorem 1.

Matrix S(P_i, L_i, ν) - 6×6 blocks

S(P_i, L_i, ν) = [αP_i - νI, 0, 0, 0, 0, 0 ] [0, -P_i, -L_i^T, -P_i, -L_i^T, 0 ] [0, -L_i, 0, -L_i, 0, L_i ] [0, -P_i, -L_i^T, -P_i, -L_i^T, 0 ] [0, -L_i, 0, -L_i, 0, L_i ] [0, 0, L_i^T, 0, L_i^T, P_i ]
def build_S_matrix(P, L, nu, alpha, n, m): 
"""
Build 6-block S matrix.

Block dimensions: [n, n, m, n, m, n] = [4, 4, 2, 4, 2, 4] = 20 total

Args:
    P: (n, n) symmetric positive definite
    L: (m, n) = K @ P
    nu: scalar slack
    alpha: decay rate (0.98)
    
Returns:
    S: (20, 20) symmetric matrix
"""
total_dim = 4*n + 2*m  # 20 for our case
S = np.zeros((total_dim, total_dim))

# Block indices: [0, 4, 8, 10, 14, 16, 20]
idx = np.cumsum([0, n, n, m, n, m, n])

# (1,1) block: αP - νI
S[idx[0]:idx[1], idx[0]:idx[1]] = alpha * P - nu * np.eye(n)

# Fill remaining blocks according to the pattern...
# (2,2): -P
S[idx[1]:idx[2], idx[1]:idx[2]] = -P
# (2,3): -L^T
S[idx[1]:idx[2], idx[2]:idx[3]] = -L.T
# ... continue for all 36 block positions

return S

Matrix Ñ₁ (Data Constraint)

N₁ = S_W M_W S_W^T where: S_W^T = [I, H_i^+^T; 0, -H_i^T; 0, -Ξ_i^T; 0, 0; 0, 0] M_W = diag(β_{i-1} I, -I) Ñ₁ = [N₁, 0; 0, 0] (padded to 6-block size)

Matrix Ñ₂ (Jacobian Variation Constraint)

N₂ = S_Δ^T M_Δ S_Δ where: S_Δ = [I, 0, 0, 0, 0; 0, 0, 0, I, 0; 0, 0, 0, 0, I] M_Δ = diag(C²T̃²I_n, -I_n, -I_m) Ñ₂ = [N₂, 0; 0, 0] (padded to 6-block size)

Purpose

Solve the main SDP (Eq. 22) to get (P_i, K_i).

Optimization Problem

maximize log det(P_i) subject to P_i ≻ 0 λ₁ ≥ 0, λ₂ ≥ 0 S(P_i, L_i, ν) - λ₁ Ñ₁ - λ₂ Ñ₂ ⪰ 0 P_i ⪰ P_min,i [R_max,i, L_i; L_i^T, P_i] ⪰ 0
def solve_funnel_sdp(H, H_plus, Xi, beta, P_min, R_max, alpha, C, T_tilde, P_prev=None, mu=1.02): 
"""
Solve the main SDP for segment i.

Args:
    H, H_plus, Xi: data matrices from segment i-1
    beta: disturbance bound from segment i-1
    P_min, R_max: feasibility ellipsoid bounds for segment i
    alpha: decay rate (0.98)
    C, T_tilde: Jacobian variation constants
    P_prev: previous P matrix (for P_i ⪯ μ P_{i-1} constraint)
    mu: growth bound (1.02)
    
Returns:
    P_i: (n, n) optimal P matrix
    K_i: (m, n) feedback gain = L_i @ inv(P_i)
    success: bool
"""
n, m = 4, 2

# Decision variables
P = cp.Variable((n, n), symmetric=True)
L = cp.Variable((m, n))
lambda1 = cp.Variable(nonneg=True)
lambda2 = cp.Variable(nonneg=True)
nu = cp.Variable(pos=True)

constraints = [
    P >> 0,
    P >> P_min,
]

# Schur complement for R_max constraint
schur_block = cp.bmat([[R_max, L], [L.T, P]])
constraints.append(schur_block >> 0)

# Main stability LMI: S - λ₁ Ñ₁ - λ₂ Ñ₂ ≥ 0
# Build S, N1_tilde, N2_tilde with CVXPY expressions
# (Requires careful construction - see lmi_matrices.py)

# Cross-segment growth (if not first segment)
if P_prev is not None:
    constraints.append(P << mu * P_prev)

# Objective: maximize log det(P)
objective = cp.Maximize(cp.log_det(P))

prob = cp.Problem(objective, constraints)
prob.solve(solver=cp.SCS, verbose=False)

if prob.status in ['optimal', 'optimal_inaccurate']:
    P_opt = P.value
    L_opt = L.value
    K_opt = L_opt @ np.linalg.inv(P_opt)
    return P_opt, K_opt, True
else:
    return None, None, False
CVXPY Implementation Note: The main challenge is constructing S(P, L, ν) - λ₁ Ñ₁ - λ₂ Ñ₂ as a valid CVXPY expression. You need to:
  1. Build S block-by-block using cp.bmat()
  2. Ñ₁ depends on data (H, H_plus, Xi, beta) - these are constants
  3. Ñ₂ depends only on (C, T_tilde) - also constants
  4. The combination with λ₁, λ₂ is affine in the decision variables
Solver Choice: Paper uses SCS. Consider also MOSEK (faster, commercial) or CVXOPT.

        controller.end_of_segment_update(k + 1)
